\section{动词}


\subsection{３类动词(カ変／サ変動詞)}

\begin{enumerate}
    \item \furigana{来}{く}： くる　$\rightarrow$ きます
    \item 名詞 ＋　する　動詞：　する　$\rightarrow$ します　
\end{enumerate}


\subsection{２类动词(上一段/下一段动词)}

\begin{enumerate}
    \item い段+る
    \item え段+る
\end{enumerate}

把词尾“る”去掉后加“ます”


\subsection{１类动词}

う段 $\rightarrow$ い段 + ます

\begin{enumerate}
    \item 结尾是う段非る的动词
    \item あ段+る、う段+る、お段+る
    \item 特例：
    \begin{enumerate}
        \item \furigana{知}{し}る $\rightarrow$ 知ります
        \item \furigana{焦}{あせ}る $\rightarrow$ 焦ります
        \item \furigana{要}{い}る $\rightarrow$ 要ります
        \item \furigana{帰}{かえ}る $\rightarrow$ 帰ります
        \item \furigana{返}{かえ}る $\rightarrow$ 返ります
        \item \furigana{限}{かぎ}る $\rightarrow$ 限ります
        \item \furigana{切}{き}る $\rightarrow$ 切ります
        \item \furigana{蹴}{け}る $\rightarrow$ 蹴ります
        \item \furigana{喋}{しゃべ}る $\rightarrow$ 喋ります
        \item \furigana{滑}{すべ}る $\rightarrow$ 滑ります
        \item \furigana{散}{ち}る $\rightarrow$ 散ります
        \item \furigana{入}{はい}る $\rightarrow$ 入ります
        \item \furigana{走}{はし}る $\rightarrow$ 走ります
        \item \furigana{減}{へ}る $\rightarrow$ 減ります
    \end{enumerate}
\end{enumerate}
